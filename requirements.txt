# --- Core backend (unchanged from the CLI's own requirements) ---

                        # If you build a newer librtlsdr from source
                        # (osmocom/rtl-sdr), you can bump this to >=0.3.0.

requests>=2.31.0       # TLE downloads (ISS tracking), AIS feed HTTP fallback
skyfield>=1.47         # Orbital mechanics / TLE propagation for ISS tracking
numpy>=1.26.0          # FFT / signal processing for waterfall, SSTV, Morse
scipy>=1.11.0           # Hilbert transform (SSTV continuous frequency tracking)
sounddevice>=0.4.6      # Microphone input fallback for SSTV/Morse decoding
pyModeS<3,>=2.13       # ADS-B Mode S message decoding. Pinned below v3:
                        # v3 removed the pms.adsb.* function-per-field API
                        # this app's ADS-B decoding is built on.
