# PAR AVION GUI — Python dependencies
# Installed automatically by install.sh into ./venv (no leading dot).
# Manual install: pip install -r requirements.txt   (inside an activated venv)
#
# Nothing here needs an account, a licence key or an API key.

# --- GUI ---
customtkinter>=5.2.0     # Modern Tkinter-based GUI framework
psutil>=5.9.0            # CPU/RAM telemetry widget
Pillow>=10.0.0           # Waterfall image rendering and SSTV image preview

# --- Signal processing ---
numpy>=1.26.0            # FFT, spectrum, SSTV and Morse decoding
scipy>=1.11.0            # Hilbert transform for SSTV frequency tracking.
                         # Optional: sstv.py falls back to a pure-numpy
                         # analytic signal if scipy is unavailable.

# --- Radio hardware ---
pyrtlsdr==0.2.93         # RTL-SDR interface — pinned: apt's librtlsdr
                         # (2.0.2 on Kali/Raspberry Pi OS) predates symbols
                         # pyrtlsdr>=0.3.0 expects (rtlsdr_set_dithering).
                         # Building librtlsdr from source (osmocom/rtl-sdr)
                         # lets you raise this to >=0.3.0.
pyModeS<3,>=2.13         # ADS-B Mode S decoding. Pinned below v3: v3
                         # removed the pms.adsb.* API this depends on.
sounddevice>=0.4.6       # Soundcard input for SSTV/Morse from an external rig

# --- Satellite tracking ---
sgp4>=2.21               # SGP4 orbit propagation, pure Python with an
                         # optional C accelerator. No data files, no
                         # downloads: the ISS tracker runs offline.
                         # (Replaces skyfield, which pulled in ephemeris
                         # files and was far heavier for what is needed.)

# --- Optional conveniences ---
requests>=2.31.0         # Only used to refresh orbital elements. Everything
                         # works without network access; see data/tle/.
gpsd-py3>=0.3.0          # GPSD client for a GPS dongle
pyserial>=3.5            # Serial fallback for /dev/ttyUSB* GPS devices

# Notes:
#  * 'tkinter' is part of the standard library but Debian/Kali/Raspberry
#    Pi OS package it separately as 'python3-tk'. It cannot come from pip
#    — install.sh installs it with apt, and creates the venv with
#    --system-site-packages so the venv can see it.
#  * The ADS-B decoder itself (dump1090) is vendored as C source in
#    third_party/dump1090 and built by install.sh — it is not a pip
#    package and does not need to be installed separately.
